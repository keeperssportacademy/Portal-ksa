function doPost(e) { try { var data = JSON.parse(e.postData.contents); var accion = data.accion; var ss = SpreadsheetApp.getActiveSpreadsheet(); if (accion === 'crearInscripcion') { var hoja = ss.getSheetByName("Atletas") || ss.insertSheet("Atletas"); hoja.appendRow([ new Date(), data.datos.nombreAtleta, data.datos.cedulaAtleta, data.datos.fechaNac, data.datos.nombreRep, data.datos.telRep, data.datos.correoRep, data.datos.plan ]); return ContentService.createTextOutput(JSON.stringify({status: "success"})).setMimeType(ContentService.MimeType.JSON); } if (accion === 'crearReserva') { var hoja = ss.getSheetByName("Reservas") || ss.insertSheet("Reservas"); hoja.appendRow([ new Date(), data.datos.atleta, data.datos.tipo, data.datos.fecha, data.datos.turno, data.datos.motivo, data.datos.referencia ]); return ContentService.createTextOutput(JSON.stringify({status: "success"})).setMimeType(ContentService.MimeType.JSON); } if (accion === 'crearPago') { var hoja = ss.getSheetByName("Pagos") || ss.insertSheet("Pagos"); hoja.appendRow([ new Date(), data.datos.atleta, data.datos.cedula, data.datos.concepto, data.datos.tipoTransaccion, data.datos.mes, data.datos.dia, data.datos.montoUSD, data.datos.montoBS, data.datos.metodo, data.datos.referencia ]); return ContentService.createTextOutput(JSON.stringify({status: "success"})).setMimeType(ContentService.MimeType.JSON); } return ContentService.createTextOutput(JSON.stringify({status: "error", message: "Acción no válida"})).setMimeType(ContentService.MimeType.JSON); } catch (error) { return ContentService.createTextOutput(JSON.stringify({status: "error", message: error.toString()})).setMimeType(ContentService.MimeType.JSON); } }
